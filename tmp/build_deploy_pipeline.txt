<#

.PARAMETER BuildType
    Build configuration (Debug, Release, Master)
    
.PARAMETER Version

.PARAMETER SkipTests
    Skip unit tests during build process
    
.PARAMETER CreateInstaller

.PARAMETER Deploy

.NOTES
    Author: Unity Tecmo Bowl Project

    Requires: Unity Hub, Visual Studio 2022, NSIS (for Windows installer)
#>

param(
    [ValidateSet("Windows", "Mac", "Linux", "WebGL", "All")]
    [string]$BuildTarget = "Windows",
    
    [ValidateSet("Debug", "Release", "Master")]
    [string]$BuildType = "Release",
    
    [string]$Version = "",
    
    [switch]$SkipTests,
    [switch]$CreateInstaller,
    [switch]$Deploy,
    [switch]$CleanBuild,
    
    [string]$ProjectPath = "C:\GameDev\TecmoBowlRemake\TecmoBowlUnity",
    [string]$BuildOutputPath = "C:\GameDev\TecmoBowlRemake\Build",
    [string]$UnityVersion = "2022.3.45f1"
)

# Global variables
$LogPath = "$env:TEMP\TecmoBowl_Build_$(Get-Date -Format 'yyyyMMdd_HHmmss').log"
$StartTime = Get-Date
$BuildSuccess = $false
$CurrentVersion = ""

# Build statistics
$BuildStats = @{
    StartTime = $StartTime
    EndTime = $null
    Duration = $null
    LinesOfCode = 0
    FilesBuilt = 0
    TestsPassed = 0
    TestsFailed = 0
    BuildSize = 0
    Platform = $BuildTarget
    Configuration = $BuildType
}

function Write-Log {
    param([string]$Message, [string]$Level = "INFO")
    $timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
    $logEntry = "[$timestamp] [$Level] $Message"
    
    $color = switch($Level) {
        "ERROR" { "Red" }
        "WARN" { "Yellow" }
        "SUCCESS" { "Green" }
        "BUILD" { "Cyan" }
        default { "White" }
    }
    
    Write-Host $logEntry -ForegroundColor $color
    Add-Content -Path $LogPath -Value $logEntry
}

function Initialize-BuildEnvironment {
    Write-Log "Initializing Tecmo Bowl build environment..." "BUILD"
    
    # Verify required tools
    if (!(Test-UnityInstallation)) {
        throw "Unity $UnityVersion not found. Please run setup script first."
    }
    
    if (!(Test-Path $ProjectPath)) {
        throw "Unity project not found at: $ProjectPath"
    }
    
    # Create build output directories
    $script:BuildOutputPath = $BuildOutputPath
    if (!(Test-Path $BuildOutputPath)) {
        New-Item -ItemType Directory -Path $BuildOutputPath -Force | Out-Null
        Write-Log "Created build output directory: $BuildOutputPath"
    }
    
    # Clean previous builds if requested
    if ($CleanBuild) {
        Write-Log "Cleaning previous builds..." "BUILD"
        Remove-Item "$BuildOutputPath\*" -Recurse -Force -ErrorAction SilentlyContinue
    }
    
    # Set version
    $script:CurrentVersion = Get-NextVersion
    Write-Log "Building version: $CurrentVersion" "BUILD"
    
    Write-Log "Build environment initialized successfully" "SUCCESS"
}

function Test-UnityInstallation {
    $unityPath = Get-UnityPath
    if (Test-Path $unityPath) {
        Write-Log "Unity found at: $unityPath" "SUCCESS"
        return $true
    }
    
    Write-Log "Unity $UnityVersion not found" "ERROR"
    return $false
}

function Get-UnityPath {
    $hubPath = "${env:ProgramFiles}\Unity\Hub\Editor\$UnityVersion\Editor\Unity.exe"
    if (Test-Path $hubPath) {
        return $hubPath
    }
    
    # Alternative installation paths
    $altPaths = @(
        "${env:ProgramFiles}\Unity\Editor\Unity.exe",
        "${env:ProgramFiles(x86)}\Unity\Editor\Unity.exe"
    )
    
    foreach ($path in $altPaths) {
        if (Test-Path $path) {
            return $path
        }
    }
    
    return $null
}

function Get-NextVersion {
    if ($Version) {
        return $Version
    }
    
    # Read current version from project settings
    $versionFile = "$ProjectPath\ProjectSettings\ProjectSettings.asset"
    if (Test-Path $versionFile) {
        $content = Get-Content $versionFile
        $versionLine = $content | Where-Object { $_ -match "bundleVersion:" }
        if ($versionLine) {
            $currentVersion = ($versionLine -split ":")[1].Trim()
            Write-Log "Current version: $currentVersion"
            
            # Auto-increment patch version
            if ($currentVersion -match "(\d+)\.(\d+)\.(\d+)") {
                $major = [int]$matches[1]
                $minor = [int]$matches[2]
                $patch = [int]$matches[3] + 1
                return "$major.$minor.$patch"
            }
        }
    }
    
    # Default version if none found
    return "1.0.0"
}

function Update-ProjectVersion {
    param([string]$NewVersion)
    
    Write-Log "Updating project version to: $NewVersion" "BUILD"
    
    # Update ProjectSettings.asset
    $settingsFile = "$ProjectPath\ProjectSettings\ProjectSettings.asset"
    if (Test-Path $settingsFile) {
        $content = Get-Content $settingsFile
        $content = $content -replace "(bundleVersion:).*", "`$1 $NewVersion"
        Set-Content $settingsFile $content
        Write-Log "Updated ProjectSettings.asset with new version"
    }
    
    # Update version in C# code
    $versionFile = "$ProjectPath\Assets\Scripts\GameVersion.cs"
    $versionCode = @"
// Auto-generated by build pipeline
using UnityEngine;

public static class GameVersion
{
    public const string VERSION = "$NewVersion";
    public const string BUILD_DATE = "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')";
    public const string BUILD_TARGET = "$BuildTarget";
    public const string BUILD_TYPE = "$BuildType";
    
    public static string GetFullVersionString()
    {
        return `$"Tecmo Bowl Remake v{VERSION} ({BUILD_TYPE}) - Built {BUILD_DATE}";
    }
}
"@
    
    # Ensure Scripts directory exists
    $scriptsDir = "$ProjectPath\Assets\Scripts"
    if (!(Test-Path $scriptsDir)) {
        New-Item -ItemType Directory -Path $scriptsDir -Force | Out-Null
    }
    
    Set-Content $versionFile $versionCode
    Write-Log "Generated GameVersion.cs with build information"
}

function Invoke-UnityTests {
    if ($SkipTests) {
        Write-Log "Skipping tests per user request" "WARN"
        return $true
    }
    
    Write-Log "Running Unity tests..." "BUILD"
    
    $unityPath = Get-UnityPath
    $testResultsPath = "$BuildOutputPath\test-results.xml"
    
    $testArgs = @(
        "-batchmode"
        "-quit"
        "-projectPath", "`"$ProjectPath`""
        "-runTests"
        "-testPlatform", "playmode"
        "-testResults", "`"$testResultsPath`""
        "-logFile", "`"$BuildOutputPath\unity-test.log`""
    )
    
    try {
        $process = Start-Process -FilePath $unityPath -ArgumentList $testArgs -PassThru -Wait
        
        if ($process.ExitCode -eq 0) {
            Write-Log "Unity tests completed successfully" "SUCCESS"
            
            # Parse test results
            if (Test-Path $testResultsPath) {
                Parse-TestResults $testResultsPath
            }
            
            return $true
        } else {
            Write-Log "Unity tests failed with exit code: $($process.ExitCode)" "ERROR"
            return $false
        }
    } catch {
        Write-Log "Error running Unity tests: $($_.Exception.Message)" "ERROR"
        return $false
    }
}

function Parse-TestResults {
    param([string]$ResultsPath)
    
    try {
        [xml]$results = Get-Content $ResultsPath
        $testSuite = $results.'test-run'
        
        $BuildStats.TestsPassed = [int]$testSuite.passed
        $BuildStats.TestsFailed = [int]$testSuite.failed
        
        Write-Log "Test Results: $($BuildStats.TestsPassed) passed, $($BuildStats.TestsFailed) failed" "BUILD"
        
        if ($BuildStats.TestsFailed -gt 0) {
            Write-Log "Some tests failed - check test-results.xml for details" "WARN"
        }
    } catch {
        Write-Log "Error parsing test results: $($_.Exception.Message)" "WARN"
    }
}

function Invoke-UnityBuild {
    param([string]$Platform)
    
    Write-Log "Building for platform: $Platform" "BUILD"
    
    $unityPath = Get-UnityPath
    $buildPath = "$BuildOutputPath\$Platform"
    
    # Create platform-specific build directory
    if (!(Test-Path $buildPath)) {
        New-Item -ItemType Directory -Path $buildPath -Force | Out-Null
    }
    
    # Determine Unity build target
    $unityTarget = switch ($Platform) {
        "Windows" { "Win64" }
        "Mac" { "OSXUniversal" }
        "Linux" { "Linux64" }
        "WebGL" { "WebGL" }
        default { "Win64" }
    }
    
    # Determine executable name
    $exeName = switch ($Platform) {
        "Windows" { "TecmoBowlRemake.exe" }
        "Mac" { "TecmoBowlRemake.app" }
        "Linux" { "TecmoBowlRemake" }
        "WebGL" { "index.html" }
        default { "TecmoBowlRemake.exe" }
    }
    
    $outputPath = "$buildPath\$exeName"
    
    # Unity build arguments
    $buildArgs = @(
        "-batchmode"
        "-quit"
        "-projectPath", "`"$ProjectPath`""
        "-buildTarget", $unityTarget
        "-executeMethod", "TecmoBowlBuildScript.BuildGame"
        "-logFile", "`"$BuildOutputPath\unity-build-$Platform.log`""
        "-buildPath", "`"$outputPath`""
        "-buildType", $BuildType
    )
    
    try {
        Write-Log "Starting Unity build process..." "BUILD"
        $process = Start-Process -FilePath $unityPath -ArgumentList $buildArgs -PassThru -Wait
        
        if ($process.ExitCode -eq 0 -and (Test-Path $outputPath)) {
            Write-Log "Unity build completed successfully for $Platform" "SUCCESS"
            
            # Calculate build size
            if ($Platform -eq "WebGL") {
                $buildSize = (Get-ChildItem $buildPath -Recurse | Measure-Object -Property Length -Sum).Sum
            } else {
                $buildSize = (Get-Item $outputPath).Length
            }
            
            $BuildStats.BuildSize += $buildSize
            $sizeInMB = [math]::Round($buildSize / 1MB, 2)
            Write-Log "Build size for $Platform`: $sizeInMB MB" "BUILD"
            
            return $true
        } else {
            Write-Log "Unity build failed for $Platform with exit code: $($process.ExitCode)" "ERROR"
            return $false
        }
    } catch {
        Write-Log "Error during Unity build: $($_.Exception.Message)" "ERROR"
        return $false
    }
}

function Create-BuildScript {
    # Create Unity build script if it doesn't exist
    $buildScriptPath = "$ProjectPath\Assets\Editor\TecmoBowlBuildScript.cs"
    
    # Ensure Editor directory exists
    $editorDir = "$ProjectPath\Assets\Editor"
    if (!(Test-Path $editorDir)) {
        New-Item -ItemType Directory -Path $editorDir -Force | Out-Null
    }
    
    $buildScriptContent = @'
using UnityEngine;
using UnityEditor;
using System;

public class TecmoBowlBuildScript
{
    public static void BuildGame()
    {
        string[] args = Environment.GetCommandLineArgs();
        string buildPath = GetArgValue(args, "-buildPath");
        string buildType = GetArgValue(args, "-buildType");
        
        if (string.IsNullOrEmpty(buildPath))
        {
            Debug.LogError("Build path not specified");
            EditorApplication.Exit(1);
            return;
        }
        
        // Configure build settings based on build type
        BuildPlayerOptions buildOptions = new BuildPlayerOptions();
        buildOptions.scenes = GetEnabledScenes();
        buildOptions.locationPathName = buildPath;
        buildOptions.target = EditorUserBuildSettings.activeBuildTarget;
        
        // Set build options based on type
        switch (buildType?.ToLower())
        {
            case "debug":
                buildOptions.options = BuildOptions.Development | BuildOptions.AllowDebugging;
                break;
            case "release":
                buildOptions.options = BuildOptions.None;
                break;
            case "master":
                buildOptions.options = BuildOptions.None;
                PlayerSettings.stripEngineCode = true;
                break;
            default:
                buildOptions.options = BuildOptions.None;
                break;
        }
        
        Debug.Log($"Building to: {buildPath} with options: {buildOptions.options}");
        
        var report = BuildPipeline.BuildPlayer(buildOptions);
        
        if (report.summary.result == UnityEditor.Build.Reporting.BuildResult.Succeeded)
        {
            Debug.Log($"Build succeeded: {report.summary.outputPath}");
            EditorApplication.Exit(0);
        }
        else
        {
            Debug.LogError($"Build failed: {report.summary.result}");
            EditorApplication.Exit(1);
        }
    }
    
    private static string GetArgValue(string[] args, string argName)
    {
        for (int i = 0; i < args.Length - 1; i++)
        {
            if (args[i] == argName)
            {
                return args[i + 1];
            }
        }
        return null;
    }
    
    private static string[] GetEnabledScenes()
    {
        var scenes = new List<string>();
        foreach (var scene in EditorBuildSettings.scenes)
        {
            if (scene.enabled)
            {
                scenes.Add(scene.path);
            }
        }
        return scenes.ToArray();
    }
}
'@
    
    Set-Content $buildScriptPath $buildScriptContent
    Write-Log "Created Unity build script at: $buildScriptPath"
}

    }

function Create-WindowsInstaller {
    # Check for NSIS installation
    $nsisPath = "${env:ProgramFiles(x86)}\NSIS\makensis.exe"
    if (!(Test-Path $nsisPath)) {
        $nsisPath = "${env:ProgramFiles}\NSIS\makensis.exe"
    }
    
    if (!(Test-Path $nsisPath)) {
        Write-Log "NSIS not found. Installing via Chocolatey..." "BUILD"
        try {
            choco install nsis -y
            $nsisPath = "${env:ProgramFiles(x86)}\NSIS\makensis.exe"
        } catch {
            Write-Log "Failed to install NSIS: $($_.Exception.Message)" "ERROR"
            return $false
        }
    }
    
    # Create NSIS installer script
    $installerScript = Create-NSISScript
    $scriptPath = "$BuildOutputPath\TecmoBowl_Installer.nsi"
    Set-Content $scriptPath $installerScript
    
    try {
        # Generate installer
        $process = Start-Process -FilePath $nsisPath -ArgumentList "`"$scriptPath`"" -PassThru -Wait
        
        if ($process.ExitCode -eq 0) {
            Write-Log "Windows installer created successfully" "SUCCESS"
            return $true
        } else {
            Write-Log "Installer creation failed with exit code: $($process.ExitCode)" "ERROR"
            return $false
        }
    } catch {
        Write-Log "Error creating Windows installer: $($_.Exception.Message)" "ERROR"
        return $false
    }
}

function Invoke-QualityGates {
    Write-Log "Running quality gates..." "BUILD"
    
    $allPassed = $true
    
    # Code analysis (simplified)
    Write-Log "Running code analysis..." "BUILD"
    
    # Check for TODO/FIXME comments
    $scriptsPath = "$ProjectPath\Assets\Scripts"
    if (Test-Path $scriptsPath) {
        $todoItems = Select-String -Path "$scriptsPath\*.cs" -Pattern "TODO|FIXME|HACK" -Recurse
        if ($todoItems.Count -gt 0) {
            Write-Log "Found $($todoItems.Count) TODO/FIXME items in code" "WARN"
        }
    }
    
    # Performance checks
    Write-Log "Checking build performance metrics..." "BUILD"
    
    # Build size check
    $maxSizeMB = 500 # Max 500MB for the game
    $actualSizeMB = [math]::Round($BuildStats.BuildSize / 1MB, 2)
    
    if ($actualSizeMB -gt $maxSizeMB) {
        Write-Log "Build size ($actualSizeMB MB) exceeds maximum ($maxSizeMB MB)" "ERROR"
        $allPassed = $false
    } else {
        Write-Log "Build size check passed: $actualSizeMB MB" "SUCCESS"
    }
    
    # Test coverage check
    if ($BuildStats.TestsFailed -gt 0) {
        Write-Log "Some tests failed - quality gate not passed" "ERROR"
        $allPassed = $false
    }
    
    if ($allPassed) {
        Write-Log "All quality gates passed" "SUCCESS"
    } else {
        Write-Log "Some quality gates failed" "ERROR"
    }
    
    return $allPassed
}

function Create-ReleasePackage {

    }
    
    # Copy documentation
    $docsPath = "$ProjectPath\..\Documentation"
    if (Test-Path $docsPath) {
        Copy-Item $docsPath "$releaseDir\Documentation" -Recurse -Force
    }
    
    # Create release notes
    $releaseNotes = Create-ReleaseNotes
    Set-Content "$releaseDir\RELEASE_NOTES.txt" $releaseNotes
    
    # Create ZIP package

        Write-Log "Release ZIP created: $zipPath"
        return $zipPath
    } catch {
        Write-Log "Failed to create release ZIP: $($_.Exception.Message)" "ERROR"
        return $null
    }

function Main {
    try {
        Write-Log "Starting Tecmo Bowl Remake Build Pipeline" "BUILD"
        Write-Log "Build configuration: $BuildType for $BuildTarget" "BUILD"
        
        # Initialize environment
        Initialize-BuildEnvironment
        
        # Update version
        Update-ProjectVersion $CurrentVersion
        
        # Create build script
        Create-BuildScript
        
        # Optimize assets
        Optimize-BuildAssets
        
        # Run tests
        if (!(Invoke-UnityTests)) {
            throw "Tests failed"
        }
        
        # Build for specified platforms
        $platformsTouild = if ($BuildTarget -eq "All") { 
            @("Windows", "Mac", "Linux", "WebGL") 
        } else { 
            @($BuildTarget) 
        }
        
        foreach ($platform in $platformsTouild) {
            if (!(Invoke-UnityBuild $platform)) {
                throw "Build failed for platform: $platform"
            }
            
            # Create installer
            if (!(Create-Installer $platform)) {
                Write-Log "Installer creation failed for $platform" "WARN"
            }
        }
        
        # Quality gates
        if (!(Invoke-QualityGates)) {
            Write-Log "Quality gates failed, but continuing build..." "WARN"
        }
        
        # Deploy if requested
        if ($Deploy) {
            if (!(Invoke-Deployment)) {
                throw "Deployment failed"
            }
        }
        
        $script:BuildSuccess = $true
        return 0
        
    } catch {
        Write-Log "Build pipeline failed: $($_.Exception.Message)" "ERROR"
        $script:BuildSuccess = $false
        return 1
        
    } finally {
        Write-BuildSummary
    }
}

# Execute the build pipeline
exit (Main)
